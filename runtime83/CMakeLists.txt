cmake_minimum_required(VERSION 3.16)
project(d1_adapter83 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_executable(probe83 probe83.cpp)
target_include_directories(probe83 PRIVATE "${REPOSITORY_ROOT}")
target_compile_options(probe83 PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_link_libraries(probe83 OpenSSL::Crypto Threads::Threads)
option(ADAPTER83_WITH_ROS "Build the opt-in synthetic DDS entry point" OFF)
if(ADAPTER83_WITH_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(std_msgs REQUIRED)
  add_executable(ros_probe83 ros_probe83.cpp)
  target_include_directories(ros_probe83 PRIVATE "${REPOSITORY_ROOT}")
  target_compile_options(ros_probe83 PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  target_link_libraries(ros_probe83 OpenSSL::Crypto Threads::Threads)
  ament_target_dependencies(ros_probe83 rclcpp std_msgs)
endif()
