cmake_minimum_required(VERSION 3.16)
project(d1_runtime03 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(mujoco 3.3.0 EXACT REQUIRED CONFIG)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime PATHS /usr/lib REQUIRED NO_DEFAULT_PATH)
add_executable(d1_runtime03 src/main.cpp)
target_include_directories(d1_runtime03 PRIVATE include /usr/include)
target_compile_options(d1_runtime03 PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_link_libraries(d1_runtime03 mujoco::mujoco ${ONNXRUNTIME_LIBRARY} OpenSSL::Crypto)
ament_target_dependencies(d1_runtime03 rclcpp geometry_msgs nav_msgs std_msgs)
set_target_properties(d1_runtime03 PROPERTIES BUILD_RPATH "/opt/ros/humble/lib;/home/lyh/ddt_ros2_ws/install/mujoco/lib")
install(TARGETS d1_runtime03 DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_package()
